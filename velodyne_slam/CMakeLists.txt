cmake_minimum_required(VERSION 3.8)
project(velodyne_slam)
cmake_policy(SET CMP0074 NEW)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(PCL REQUIRED COMPONENTS common filters io registration kdtree)
find_package(Eigen3 REQUIRED)
find_package(Threads REQUIRED)

add_library(cloud_preprocessor src/cloud_preprocessor.cpp)
target_compile_features(cloud_preprocessor PUBLIC cxx_std_17)
target_include_directories(cloud_preprocessor PUBLIC include ${PCL_INCLUDE_DIRS})
target_link_libraries(cloud_preprocessor ${PCL_LIBRARIES})

add_executable(slam_node src/slam_node.cpp)
target_compile_features(slam_node PUBLIC cxx_std_17)
target_link_libraries(slam_node cloud_preprocessor)
ament_target_dependencies(slam_node rclcpp sensor_msgs std_msgs pcl_conversions)

add_library(ndt_matcher src/ndt_matcher.cpp)
target_compile_features(ndt_matcher PUBLIC cxx_std_17)
target_include_directories(ndt_matcher PUBLIC include ${PCL_INCLUDE_DIRS})
target_link_libraries(ndt_matcher ${PCL_LIBRARIES})
add_executable(ndt_localization_node src/ndt_localization_node.cpp)
target_link_libraries(ndt_localization_node ndt_matcher Threads::Threads)
ament_target_dependencies(ndt_localization_node rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs tf2_ros pcl_conversions)
add_executable(map_builder_node src/map_builder_node.cpp)
target_compile_features(map_builder_node PUBLIC cxx_std_17)
target_include_directories(map_builder_node PRIVATE include ${PCL_INCLUDE_DIRS})
target_link_libraries(map_builder_node ${PCL_LIBRARIES} Eigen3::Eigen)
ament_target_dependencies(map_builder_node rclcpp sensor_msgs nav_msgs std_msgs std_srvs pcl_conversions)

install(TARGETS cloud_preprocessor slam_node ndt_matcher ndt_localization_node map_builder_node
  ARCHIVE DESTINATION lib
  LIBRARY DESTINATION lib
  RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_localization_stability test/test_localization_stability.cpp)
  target_compile_features(test_localization_stability PRIVATE cxx_std_17)
  target_include_directories(test_localization_stability PRIVATE include)
  target_link_libraries(test_localization_stability Eigen3::Eigen)
  ament_add_gtest(test_initial_pose test/test_initial_pose.cpp)
  target_include_directories(test_initial_pose PRIVATE include)
  target_link_libraries(test_initial_pose Eigen3::Eigen)
  ament_add_gtest(test_ndt_matcher test/test_ndt_matcher.cpp)
  target_link_libraries(test_ndt_matcher ndt_matcher)
  ament_add_gtest(test_lidar_origin test/test_lidar_origin.cpp)
  target_compile_features(test_lidar_origin PRIVATE cxx_std_17)
  target_include_directories(test_lidar_origin PRIVATE include ${PCL_INCLUDE_DIRS})
  target_link_libraries(test_lidar_origin Eigen3::Eigen)
  ament_add_gtest(test_consecutive_free_space test/test_consecutive_free_space.cpp)
  target_include_directories(test_consecutive_free_space PRIVATE include)
  target_compile_features(test_consecutive_free_space PRIVATE cxx_std_17)
endif()
ament_package()
